8 El siguiente fragmento calcula distancia recorrida a partir de eventos GPS, pero suma distancia incluso cuando el vehículo está detenido (por ruido del sensor). Identifica el bug y propón el fix.
public double CalcularDistanciaRecorrida(List<PosicionGps> posiciones) { double distanciaTotal = 0; for (int i = 1; i < posiciones.Count; i++) { var anterior = posiciones[i - 1]; var actual = posiciones[i]; double distancia = CalcularDistanciaHaversine( anterior.Latitud, anterior.Longitud, actual.Latitud, actual.Longitud); distanciaTotal += distancia; } return distanciaTotal; }
9 La siguiente función recorre miles de registros de posiciones para calcular kilometraje total, con un problema de rendimiento evidente (ej. consulta a base de datos dentro del loop). Identifícalo y propón una mejora.
public double CalcularKilometrajeTotal(List<int> vehiculoIds) { double kilometrajeTotal = 0; foreach (var vehiculoId in vehiculoIds) { var posiciones = _dbContext.PosicionesGps .Where(p => p.VehiculoId == vehiculoId) .OrderBy(p => p.Timestamp) .ToList(); for (int i = 1; i < posiciones.Count; i++) { kilometrajeTotal += CalcularDistanciaHaversine( posiciones[i - 1].Latitud, posiciones[i - 1].Longitud, posiciones[i].Latitud, posiciones[i].Longitud); } } return kilometrajeTotal; } // Se llama con vehiculoIds de ~500 vehículos activos
